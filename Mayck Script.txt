local function connect(signal, handler)
    local c = signal:Connect(handler)
    table.insert(connections, c)
    return c
end
local function new(className, props, parent)
    local o = Instance.new(className)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end
local function corner(parent, radius)
    return new("UICorner", {CornerRadius = UDim.new(0, radius or 10)}, parent)
end
local function stroke(parent, color, transparency)
    return new("UIStroke", {Color = color or C.BORDER, Thickness = 1,
        Transparency = transparency or 0}, parent)
end
local function bindTheme(object, propertyName, token)
    object[propertyName] = C[token]
    table.insert(themeBindings, {object = object, propertyName = propertyName, token = token})
end
local function applyTheme(name)
    if not THEMES[name] then return end
    settings.Theme = name
    C = THEMES[name]
    for _, entry in ipairs(themeBindings) do
        if entry.object.Parent then
            pcall(function() entry.object[entry.propertyName] = C[entry.token] end)
        end
    end
end
local function tween(object, seconds, properties, style, direction)
    if destroyed or not object or not object.Parent then return end
    local tw = TweenService:Create(object, TweenInfo.new(seconds or 0.18,
        style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out), properties)
    tw:Play()
    return tw
end
local function textLabel(parent, value, size, bold, colorToken)
    local t = new("TextLabel", { BackgroundTransparency = 1, Text = value,
        Font = bold and Enum.Font.GothamBold or Enum.Font.Gotham,
        TextSize = size or 12, TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd, TextWrapped = false }, parent)
    bindTheme(t, "TextColor3", colorToken or "TEXT")
    return t
end
local function basicButton(parent, title, width)
    local b = new("TextButton", {Size = UDim2.fromOffset(width or 96, 34),
        Text = title, Font = Enum.Font.GothamBold, TextSize = 11,
        AutoButtonColor = false, BorderSizePixel = 0}, parent)
    bindTheme(b, "BackgroundColor3", "CARD")
    bindTheme(b, "TextColor3", "TEXT")
    corner(b, 9)
    local s = stroke(b)
    bindTheme(s, "Color", "BORDER")
    b.MouseEnter:Connect(function() if not destroyed then tween(b, 0.12, {BackgroundTransparency = 0.2}) end end)
    b.MouseLeave:Connect(function() if not destroyed then tween(b, 0.12, {BackgroundTransparency = 0}) end end)
    return b
end

V4.floatDragging = false
V4.floatIsTouch = false
V4.floatTouchInput = nil
V4.floatStart = nil
V4.floatCenterStart = nil
V4.floatLastPosition = nil
V4.floatMoved = false
V4.floatThreshold = 6
function V4.clampFloatPosition(pixel)
    local vp = viewport()
    local half = UI.FloatButton.AbsoluteSize * 0.5
    return Vector2.new(
        math.clamp(pixel.X, half.X + 6, math.max(half.X + 6, vp.X - half.X - 6)),
        math.clamp(pixel.Y, half.Y + 6, math.max(half.Y + 6, vp.Y - half.Y - 6))
    )
end
local function applyFloatPointer(position)
    if destroyed or not V4.floatDragging or not position or not V4.floatStart then return end
    V4.floatLastPosition = position
    local delta = position - V4.floatStart
    if delta.Magnitude >= V4.floatThreshold then V4.floatMoved = true end
    if V4.floatMoved and not settings.LockFloat then
        local at = V4.clampFloatPosition(V4.floatCenterStart + delta)
        UI.FloatButton.Position = UDim2.fromOffset(at.X, at.Y)
    end
end
local function finishFloatGesture()
    if not V4.floatDragging then return end
    local wasClick = not V4.floatMoved
    V4.floatDragging = false
    V4.floatTouchInput = nil
    V4.floatStart = nil
    V4.floatCenterStart = nil
    V4.floatLastPosition = nil
    V4.floatMoved = false
    UI.floatScale.Scale = 1
    if wasClick and not destroyed then togglePanel() end
end
UI.FloatButton.InputBegan:Connect(function(input)
    if destroyed or V4.floatDragging then return end
    local kind = input.UserInputType
    if kind ~= Enum.UserInputType.MouseButton1 and kind ~= Enum.UserInputType.Touch then return end
    V4.floatDragging = true
    V4.floatIsTouch = kind == Enum.UserInputType.Touch
    V4.floatTouchInput = V4.floatIsTouch and input or nil
    V4.floatStart = V4.floatIsTouch and input.Position or UserInputService:GetMouseLocation()
    V4.floatCenterStart = UI.FloatButton.AbsolutePosition + UI.FloatButton.AbsoluteSize * 0.5
    V4.floatLastPosition = V4.floatStart
    V4.floatMoved = false
    UI.floatScale.Scale = 0.94
end)

local function dragTarget(target, handle)
    local drag=false
    local dragStart, beginPosition, currentTouch
    handle.InputBegan:Connect(function(input)
        if destroyed then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            drag=true
            currentTouch=input.UserInputType==Enum.UserInputType.Touch and input or nil
            dragStart=input.Position
            beginPosition=target.Position
        end
    end)
    connect(UserInputService.InputChanged,function(input)
        if not drag or destroyed then return end
        if (currentTouch and input==currentTouch) or
            (not currentTouch and input.UserInputType==Enum.UserInputType.MouseMovement) then
            local delta=input.Position-dragStart
            local proposed=UDim2.new(beginPosition.X.Scale,beginPosition.X.Offset+delta.X,
                beginPosition.Y.Scale,beginPosition.Y.Offset+delta.Y)
            target.Position=proposed
            if target==Root then
                local v=viewport()
                local x=target.Position.X.Scale*v.X+target.Position.X.Offset
                local y=target.Position.Y.Scale*v.Y+target.Position.Y.Offset
                x=math.clamp(x,target.AbsoluteSize.X/2+8,v.X-target.AbsoluteSize.X/2-8)
                y=math.clamp(y,target.AbsoluteSize.Y/2+8,v.Y-target.AbsoluteSize.Y/2-8)
                target.Position=UDim2.fromOffset(x,y)
            elseif target==V4.RadarHUD then
                local v=viewport()
                local x=target.Position.X.Scale*v.X+target.Position.X.Offset
                local y=target.Position.Y.Scale*v.Y+target.Position.Y.Offset
                target.Position=UDim2.fromOffset(math.clamp(x,0,math.max(0,v.X-target.AbsoluteSize.X)),
                    math.clamp(y,0,math.max(0,v.Y-target.AbsoluteSize.Y)))
            end
        end
    end)
    connect(UserInputService.InputEnded,function(input)
        if (currentTouch and input==currentTouch) or
            (not currentTouch and input.UserInputType==Enum.UserInputType.MouseButton1) then
            drag=false
            currentTouch=nil
        end
    end)
end

local sliders={}
local function slider(parent, key, title, minValue, maxValue, step)
    local row = new("Frame",{Size=UDim2.new(1,0,0,68),BackgroundTransparency=1},parent)
    local label = textLabel(row,title,12,true)
    label.Size=UDim2.new(1,-100,0,24)
    local amount = textLabel(row,tostring(settings[key]),12,true,"ACCENT")
    amount.Size=UDim2.new(0,90,0,24)
    amount.Position=UDim2.new(1,-90,0,0)
    amount.TextXAlignment=Enum.TextXAlignment.Right
    local trackBar=new("Frame", {Size=UDim2.new(1,-2,0,13), Position=UDim2.fromOffset(1,35),
        BorderSizePixel=0, Active=true},row)
    bindTheme(trackBar,"BackgroundColor3","PANEL")
    corner(trackBar,7)
    local fill = new("Frame", {BorderSizePixel=0, Size=UDim2.fromScale(0,1)},trackBar)
    bindTheme(fill,"BackgroundColor3","ACCENT")
    corner(fill,7)
    local knob=new("Frame",{Size=UDim2.fromOffset(18,18),AnchorPoint=Vector2.new(0.5,0.5),
        Position=UDim2.fromScale(0,0.5),BorderSizePixel=0},trackBar)
    bindTheme(knob,"BackgroundColor3","TEXT")
    corner(knob,9)
    local function render()
        local v=math.clamp(settings[key],minValue,maxValue)
        local f=(v-minValue)/(maxValue-minValue)
        fill.Size=UDim2.fromScale(f,1)
        knob.Position=UDim2.fromScale(f,0.5)
        amount.Text=tostring(v)
    end
    local dragging=false
    local touch=nil
    local function update(input)
        local x=math.clamp((input.Position.X-trackBar.AbsolutePosition.X)/math.max(1,trackBar.AbsoluteSize.X),0,1)
        local raw=minValue+x*(maxValue-minValue)
        settings[key]=math.clamp(math.floor(raw/step+0.5)*step,minValue,maxValue)
        render()
        if featureChanged then featureChanged(key) end
    end
    local function begin(input)
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
            dragging=true
            touch=input.UserInputType==Enum.UserInputType.Touch and input or nil
            update(input)
        end
    end
    trackBar.InputBegan:Connect(begin)
    knob.InputBegan:Connect(begin)
    connect(UserInputService.InputChanged,function(input)
        if dragging and not destroyed and ((touch and touch==input) or
            (not touch and input.UserInputType==Enum.UserInputType.MouseMovement)) then update(input) end
    end)
    connect(UserInputService.InputEnded,function(input)
        if (touch and input==touch) or (not touch and input.UserInputType==Enum.UserInputType.MouseButton1) then
            dragging=false
            touch=nil
        end
    end)
    sliders[key]=render
    render()
end

    local function matchingName(instance)
        local name=instance.Name:lower()
        local parent=instance.Parent and instance.Parent.Name:lower() or ''
        -- Avoid treating arbitrary decorations as notes. Exact game-specific names may need adding here.
        if settings.expanded and (name:find('circle',1,true) or name:find('gem',1,true) or name:find('disc',1,true) or name:find('tap',1,true)) then
            return true
        end
        return name:find('note',1,true) or name:find('sustain',1,true)
            or name:find('hold',1,true) or name:find('arrow',1,true)
            or name:find('hitobject',1,true) or parent:find('notes',1,true)
    end
    local function sustainName(instance)
        local name=instance.Name:lower()
        return name:find('hold',1,true) or name:find('long',1,true)
            or name:find('sustain',1,true) or name:find('trail',1,true)
    end
    local function findLane(x,viewWidth)
        local best,dist
        for i=1,#KEYS do
            local delta=math.abs(x/viewWidth-settings.laneXs[i])
            if not dist or delta<dist then best=i;dist=delta end
        end
        -- Perspective makes top-end x vary. Lane chosen near hit line only.
        if dist and dist<=.115 then return best end
    end
    local function register(object,midX,headY,tailY,screenWidth,now,source)
        local data=trackData[object]
        if not data then
            trackData[object]={oldY=headY,oldT=now,velocity=0,triggered=false,
                lastSeen=now,tailY=tailY,source=source}
            return
        end
        local previousY=data.oldY
        local elapsed=math.max(now-data.oldT,.001)
        local instant=(headY-previousY)/elapsed
        data.velocity=data.velocity*.3+instant*.7
        data.oldY=headY;data.oldT=now;data.lastSeen=now;data.tailY=tailY
        if data.triggered or data.velocity<settings.minSpeed then return end
        local cam=workspace.CurrentCamera
        if not cam then return end
        local lineY=cam.ViewportSize.Y*settings.lineY
        -- Positive timing brings the trigger forward; negative timing delays it.
        local target=lineY-data.velocity*settings.timingMs/1000
        local lane=findLane(midX,screenWidth)
        if not lane then return end
        local crossing=previousY<target+settings.tolerance and headY>=target-settings.tolerance
        local close=math.abs(headY-target)<=settings.tolerance
        if not crossing and not close then return end
        data.triggered=true
        -- Avoid holding one lane for two different notes at once.
        if held[lane] then
            sendUp(lane);held[lane]=nil
        end
        if sendDown(lane) then
            hits+=1
            local hold=sustainName(object) or (tailY~=nil and (headY-tailY)>65)
            held[lane]={obj=object,untilTime=now+settings.pressMs/1000,
                hold=hold,startTime=now}
            currentSource=source
            lastEvent=KEYS[lane]..(hold and ' HOLD' or ' TAP')..' ['..source..']'
        else
            dropped+=1
        end
    end
    -- Indexed note candidates: scan complete hierarchies only once, then listen for additions/removals.
    -- The detector keeps its original 16 ms cadence and matching/registration algorithm.
    local guiCandidates, worldCandidates = {}, {}
    local guiIndex, worldIndex = {}, {}
    local trackedPlayerGui
    local guiAddedConn, guiRemovingConn
    local function isGuiCandidate(obj)
        return obj:IsA('GuiObject') and matchingName(obj)
    end
    local function isWorldCandidate(obj)
        return obj:IsA('BasePart') and matchingName(obj)
    end
    local function indexCandidate(obj, list, lookup, predicate)
        if not lookup[obj] and predicate(obj) then
            list[#list+1] = obj
            lookup[obj] = #list
        end
    end
    local function forgetCandidate(obj, list, lookup)
        local at = lookup[obj]
        if not at then return end
        local last = list[#list]
        list[at] = last
        if last then lookup[last] = at end
        list[#list] = nil
        lookup[obj] = nil
        trackData[obj] = nil
    end
    local function rebuildCandidates()
        table.clear(guiCandidates);table.clear(worldCandidates)
        table.clear(guiIndex);table.clear(worldIndex)
        local pg=LP:FindFirstChildOfClass('PlayerGui')
        if pg then
            for _,obj in ipairs(pg:GetDescendants()) do
                indexCandidate(obj, guiCandidates, guiIndex, isGuiCandidate)
            end
        end
        for _,obj in ipairs(workspace:GetDescendants()) do
            indexCandidate(obj, worldCandidates, worldIndex, isWorldCandidate)
        end
    end
    local function attachPlayerGui(pg)
        if guiAddedConn then guiAddedConn:Disconnect();guiAddedConn=nil end
        if guiRemovingConn then guiRemovingConn:Disconnect();guiRemovingConn=nil end
        trackedPlayerGui=pg
        table.clear(guiCandidates);table.clear(guiIndex)
        if not pg then return end
        for _,obj in ipairs(pg:GetDescendants()) do
            indexCandidate(obj,guiCandidates,guiIndex,isGuiCandidate)
        end
        guiAddedConn=pg.DescendantAdded:Connect(function(obj)
            if alive then indexCandidate(obj,guiCandidates,guiIndex,isGuiCandidate) end
        end)
        guiRemovingConn=pg.DescendantRemoving:Connect(function(obj)
            forgetCandidate(obj,guiCandidates,guiIndex)
        end)
    end
    local function scanGUI(now,view)
        local count=0
        local pg=trackedPlayerGui
        if not pg or not pg.Parent then return count end
        for i=1,#guiCandidates do
            local obj=guiCandidates[i]
            if obj and obj:IsDescendantOf(pg) and obj.Visible then
                local p,s=obj.AbsolutePosition,obj.AbsoluteSize
                if s.X>=7 and s.Y>=7 and s.X<500 and s.Y<view.Y then
                    count+=1
                    register(obj,p.X+s.X/2,p.Y+s.Y*.88,p.Y,view.X,now,'GUI')
                end
            end
        end
        return count
    end
    local function scanWorld(now,view)
        local count=0
        local camera=workspace.CurrentCamera
        if not camera then return 0 end
        for i=1,#worldCandidates do
            local obj=worldCandidates[i]
            if obj and obj:IsDescendantOf(workspace) then
                local point,onScreen=camera:WorldToViewportPoint(obj.Position)
                if onScreen and point.Z>0 and point.Y>-60 and point.Y<view.Y+100 then
                    count+=1
                    local tailY
                    if sustainName(obj) then
                        local top=camera:WorldToViewportPoint(obj.Position+obj.CFrame.UpVector*obj.Size.Y*.5)
                        tailY=math.min(point.Y,top.Y)
                    end
                    register(obj,point.X,point.Y,tailY,view.X,now,'3D')
                end
            end
        end
        return count
    end

    local function serviceHeld(now,hitY)
        for lane,data in pairs(held) do
            if not data.hold then
                if now>=data.untilTime then sendUp(lane);held[lane]=nil end
            else
                local obj= data.obj
                local note=trackData[obj]
                -- Only release after the trailing edge has passed the hit line.
                -- If an object disappears while holding, release after a short fallback.
                if not note or now-note.lastSeen>.4 or
                    (note.tailY and note.tailY>=hitY+settings.tolerance) or
                    now-data.startTime>8 then
                    sendUp(lane);held[lane]=nil
                end
            end
        end
    end

        end

        uiClock=0

        statusBadge.TextColor3=settings.active and ACCENT or MUTED
        local holdCount=0;for _ in pairs(held) do holdCount+=1 end

            notesSeen,hits,holdCount,dropped,settings.lineY*100,settings.timingMs,settings.pressMs,
            settings.gui and 'GUI ' or '',settings.world and '3D' or '',
            (type(keypress)=='function' and type(keyrelease)=='function') and 'SIM' or 'NÃO',lastEvent)

-- Inicialização por demanda: abrir o Mayck Script nunca executa o motor de notas.
-- O motor será criado somente na primeira interação com os comandos PARK HERO.
local function ensureParkEngine()
    if destroyed then return nil end
    if UI.Park.engine then return UI.Park.engine end
    UI.ParkStatus.Text="◌ Carregando motor Park Hero..."
    local ok,result=pcall(createParkEngine)
    if not ok or not result then
        UI.ParkStatus.Text="● FALHA AO CARREGAR"
        UI.ParkDetails.Text=tostring(result)
        notify("Park Hero não iniciou: "..tostring(result),false)
        return nil
    end
    UI.Park.engine=result
    UI.ParkStatus.Text="○ PRONTO • pressione INICIAR"
    return result
end
local function parkCall(method,...)
    local engine=ensureParkEngine()
    if not engine then return end
    local ok,err=pcall(engine[method],...)
    if not ok then notify("Park Hero: "..tostring(err),false) end
end

local function loadAvatar(userId,imageLabel)
    if UI.avatarCache[userId] then imageLabel.Image=UI.avatarCache[userId] return end
    if UI.avatarPending[userId] then
        task.spawn(function()
            for _=1,45 do
                if UI.avatarCache[userId] or not UI.avatarPending[userId] then break end
                task.wait(0.12)
            end
            if imageLabel.Parent and UI.avatarCache[userId] then imageLabel.Image=UI.avatarCache[userId] end
        end)
        return
    end
    UI.avatarPending[userId]=true
    task.spawn(function()
        local ok,content=pcall(function()
            local url,ready=Players:GetUserThumbnailAsync(userId,
                Enum.ThumbnailType.HeadShot,Enum.ThumbnailSize.Size100x100)
            for _=1,5 do
                if ready or destroyed then break end
                task.wait(0.3)
                url,ready=Players:GetUserThumbnailAsync(userId,
                    Enum.ThumbnailType.HeadShot,Enum.ThumbnailSize.Size100x100)
            end
            return url
        end)
        if ok and type(content)=="string" and content~="" then UI.avatarCache[userId]=content end
        UI.avatarPending[userId]=nil
        if not destroyed and imageLabel.Parent and UI.avatarCache[userId] then
            imageLabel.Image=UI.avatarCache[userId]
        end
    end)
end

V4.radarDots={}
function V4.updateRadar()
    V4.RadarHUD.Visible=settings.Radar and not destroyed
    if not settings.Radar then return end
    local r=myRoot()
    if not r then return end
    local keep={}
    local camera=Workspace.CurrentCamera
    local forward=camera and camera.CFrame.LookVector or Vector3.new(0,0,-1)
    local flat=Vector3.new(forward.X,0,forward.Z)
    if flat.Magnitude<.01 then flat=Vector3.new(0,0,-1) else flat=flat.Unit end
    local right=Vector3.new(-flat.Z,0,flat.X)
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=Me then
            local other=p.Character and p.Character:FindFirstChild("HumanoidRootPart")
            if other then
                local offset=other.Position-r.Position
                local dx=offset:Dot(right)/settings.RadarRange
                local dy=-offset:Dot(flat)/settings.RadarRange
                if dx*dx+dy*dy<=1 then
                    keep[p]=true
                    if not V4.radarDots[p] or not V4.radarDots[p].Parent then
                        V4.radarDots[p]=new("Frame",{Size=UDim2.fromOffset(6,6),
                            AnchorPoint=Vector2.new(.5,.5),ZIndex=44,BorderSizePixel=0},V4.radarArea)
                        corner(V4.radarDots[p],4)
                    end
                    local dot=V4.radarDots[p]
                    dot.Position=UDim2.fromScale(math.clamp(.5+dx*.46,.05,.95),math.clamp(.5+dy*.46,.05,.95))
                    dot.BackgroundColor3=(p.Team~=nil and Me.Team~=nil and p.Team==Me.Team) and C.GOOD or C.ACCENT2
                end
            end
        end
    end
    for p,dot in pairs(V4.radarDots) do
        if not keep[p] then dot:Destroy() V4.radarDots[p]=nil end
    end
end

        status.Text="Indisponível: LocalScripts comuns não oferecem loadstring."
        notify("Este ambiente não permite esse carregador.",false)
        return

    UI.voiceBusy[which]=true
    if which=="Mois7" then
        UI.voiceStatus1.Text="Mois7: inicialização solicitada. Se o painel aparecer, marque PRONTO."

            local body=game:HttpGet(UI.voiceSources[which])
            assert(type(body)=="string" and #body>0,"resposta vazia")

local function flyMove(dt)
    if not settings.Fly then return end
    local hrp=myRoot()
    if not hrp then return end
    if not UI.flyVelocity or UI.flyVelocity.Parent~=hrp then startFly() end
    if not UI.flyVelocity then return end
    local cam=Workspace.CurrentCamera
    if not cam then return end
    local move=Vector3.zero
    local hum=myHumanoid()
    if hum then move=hum.MoveDirection end
    if not UserInputService.TouchEnabled then
        local forward = (UserInputService:IsKeyDown(Enum.KeyCode.W) and 1 or 0)-
            (UserInputService:IsKeyDown(Enum.KeyCode.S) and 1 or 0)
        local side = (UserInputService:IsKeyDown(Enum.KeyCode.D) and 1 or 0)-
            (UserInputService:IsKeyDown(Enum.KeyCode.A) and 1 or 0)
        local look=cam.CFrame.LookVector
        local right=cam.CFrame.RightVector
        local flatLook=Vector3.new(look.X,0,look.Z)
        local flatRight=Vector3.new(right.X,0,right.Z)
        if flatLook.Magnitude>0 then flatLook=flatLook.Unit end
        if flatRight.Magnitude>0 then flatRight=flatRight.Unit end
        if forward~=0 or side~=0 then move=flatLook*forward+flatRight*side end
    end
    local vertical=0
    if UI.flyUp or UserInputService:IsKeyDown(Enum.KeyCode.Space) then vertical+=1 end
    if UI.flyDown or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or
        UserInputService:IsKeyDown(Enum.KeyCode.RightControl) then vertical-=1 end
    local desired=move+Vector3.new(0,vertical,0)
    if desired.Magnitude>1 then desired=desired.Unit end
    local targetVelocity=desired*settings.FlySpeed
    if settings.FlySmooth then
        V4.flySmoothed=V4.flySmoothed:Lerp(targetVelocity,math.clamp(dt*settings.FlyResponsiveness,0,1))
    else
        V4.flySmoothed=targetVelocity
    end
    UI.flyVelocity.VectorVelocity=V4.flySmoothed
    local flat=Vector3.new(cam.CFrame.LookVector.X,0,cam.CFrame.LookVector.Z)
    if flat.Magnitude>0.02 then
        hrp.CFrame=CFrame.lookAt(hrp.Position,hrp.Position+flat.Unit)
    end
end

-- Dash, Bunny Hop, água, modo desempenho e Click Teleport
function V4.performDash()
    if destroyed or not settings.Dash or settings.Freecam then return end
    if os.clock()-V4.lastDash<0.9 then return end
    local r=myRoot()
    local h=myHumanoid()
    if not r or not h or h.Health<=0 then return end
    local dir=h.MoveDirection
    if dir.Magnitude<.05 then
        local cam=Workspace.CurrentCamera
        dir=cam and cam.CFrame.LookVector or r.CFrame.LookVector
    end
    dir=Vector3.new(dir.X,0,dir.Z)
    if dir.Magnitude<.1 then return end
    V4.lastDash=os.clock()
    r.AssemblyLinearVelocity=dir.Unit*settings.DashStrength+Vector3.yAxis*math.max(0,r.AssemblyLinearVelocity.Y)
end

function V4.stepWater()
    if not settings.WalkOnWater then
        if V4.waterPlatform then V4.waterPlatform:Destroy() V4.waterPlatform=nil end
        return
    end
    local r=myRoot()
    if not r then return end
    local params=RaycastParams.new()
    params.FilterType=Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances={Me.Character}
    params.IgnoreWater=false
    local hit=Workspace:Raycast(r.Position+Vector3.new(0,3,0),Vector3.new(0,-18,0),params)
    if hit and hit.Material==Enum.Material.Water then
        if not V4.waterPlatform then
            V4.waterPlatform=new("Part",{Name="Mayck_WaterFooting",Size=Vector3.new(6,.5,6),
                Anchored=true,CanCollide=true,Transparency=1},Workspace)
        end
        V4.waterPlatform.Position=Vector3.new(r.Position.X,hit.Position.Y-.45,r.Position.Z)
    elseif V4.waterPlatform then
        V4.waterPlatform:Destroy()
        V4.waterPlatform=nil
    end
end
function V4.setPerformance()
    if settings.Performance then
        V4.perfLighting={Shadows=Lighting.GlobalShadows}
        Lighting.GlobalShadows=false
        for _,desc in ipairs(Workspace:GetDescendants()) do
            if desc:IsA("ParticleEmitter") or desc:IsA("Trail") or desc:IsA("Beam") then
                -- Não mexe nos feixes próprios do ESP.
                if not desc.Name:find("MayckESP") then
                    V4.perfOriginal[desc]=desc.Enabled
                    desc.Enabled=false
                end
            end
        end
    else
        if V4.perfLighting then Lighting.GlobalShadows=V4.perfLighting.Shadows V4.perfLighting=nil end
        for inst,value in pairs(V4.perfOriginal) do
            if inst.Parent then pcall(function() inst.Enabled=value end) end
        end
        table.clear(V4.perfOriginal)
    end
end
function V4.tryClickTeleport(input, processed)
    if not settings.ClickTP or processed or destroyed or isOpen or settings.Freecam then return end
    if not (input.UserInputType==Enum.UserInputType.MouseButton1 or
        input.UserInputType==Enum.UserInputType.Touch) then return end
    if os.clock()-V4.lastClickTP<.3 then return end
    local pos=input.Position
    local tilePos=UI.FloatButton.AbsolutePosition
    local tileSize=UI.FloatButton.AbsoluteSize
    if pos.X>=tilePos.X and pos.X<=tilePos.X+tileSize.X
        and pos.Y>=tilePos.Y and pos.Y<=tilePos.Y+tileSize.Y then return end
    local r=myRoot()
    local camera=Workspace.CurrentCamera
    if not r or not camera then return end
    local hitPosition
    if input.UserInputType==Enum.UserInputType.MouseButton1 then
        -- GetMouse é suportado no cliente, mas só usamos a posição para um Raycast.
        local mouse=Me:GetMouse()
        local ray=camera:ViewportPointToRay(mouse.X,mouse.Y)
        local params=RaycastParams.new()
        params.FilterType=Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances={Me.Character}
        local hit=Workspace:Raycast(ray.Origin,ray.Direction*1200,params)
        hitPosition=hit and hit.Position
    else
        local ray=camera:ViewportPointToRay(input.Position.X,input.Position.Y)
        local params=RaycastParams.new()
        params.FilterType=Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances={Me.Character}
        local hit=Workspace:Raycast(ray.Origin,ray.Direction*1200,params)
        hitPosition=hit and hit.Position
    end
    if hitPosition then
        V4.lastClickTP=os.clock()
        table.insert(V4.tpHistory,1,r.CFrame)
        if #V4.tpHistory>12 then table.remove(V4.tpHistory) end
        r.CFrame=CFrame.new(hitPosition+Vector3.new(0,4,0))
        r.AssemblyLinearVelocity=Vector3.zero
    end
end

local function startFreecam()
    local cam=Workspace.CurrentCamera
    if not cam then return end
    if spectateTarget then spectateTarget=nil refreshPlayers() end
    UI.freecamSaved={Type=cam.CameraType,Subject=cam.CameraSubject}
    UI.fcPosition=cam.CFrame.Position
    local look=cam.CFrame.LookVector
    UI.fcPitch=math.asin(math.clamp(look.Y,-1,1))
    UI.fcYaw=math.atan2(-look.X,-look.Z)
    cam.CameraType=Enum.CameraType.Scriptable
end
local function stepFreecam(dt)
    if not settings.Freecam then return end
    local cam=Workspace.CurrentCamera
    if not cam then return end
    if cam.CameraType~=Enum.CameraType.Scriptable then cam.CameraType=Enum.CameraType.Scriptable end
    if not UI.fcPosition then UI.fcPosition=cam.CFrame.Position end
    local rotation=CFrame.fromOrientation(UI.fcPitch,UI.fcYaw,0)
    local velocity=Vector3.zero
    local function held(k) return UserInputService:IsKeyDown(k) end
    if held(Enum.KeyCode.W) then velocity+=rotation.LookVector end
    if held(Enum.KeyCode.S) then velocity-=rotation.LookVector end
    if held(Enum.KeyCode.D) then velocity+=rotation.RightVector end
    if held(Enum.KeyCode.A) then velocity-=rotation.RightVector end
    if held(Enum.KeyCode.Q) then velocity+=Vector3.yAxis end
    if held(Enum.KeyCode.LeftControl) or held(Enum.KeyCode.RightControl) then velocity-=Vector3.yAxis end
    if velocity.Magnitude>1 then velocity=velocity.Unit end
    UI.fcPosition+=velocity*settings.FreecamSpeed*dt
    cam.CFrame=CFrame.new(UI.fcPosition)*rotation
end

UI.espUnready ={}
local function safeDestroy(inst)
    if inst then pcall(function() inst:Destroy() end) end
end
local function clearESPPlayer(p)
    local obj=espObjects[p]
    if not obj then return end
    for _,key in ipairs({"Highlight","Billboard","Beam","TargetAttachment","Box"}) do
        safeDestroy(obj[key])
    end
    if obj.Skeleton then
        for _,item in ipairs(obj.Skeleton) do
            safeDestroy(item.Adornment)
            safeDestroy(item.Beam)
        end
    end
    espObjects[p]=nil
end
local function clearESP()
    for p in pairs(espObjects) do clearESPPlayer(p) end
    safeDestroy(UI.espOrigin)
    UI.espOrigin=nil
    UI.espOriginRoot=nil
end
local function originAttachment()
    local root=myRoot()
    if not root then return nil end
    if root~=UI.espOriginRoot or not UI.espOrigin or not UI.espOrigin.Parent then
        safeDestroy(UI.espOrigin)
        UI.espOrigin=new("Attachment",{Name="MayckESP_Origin"},root)
        UI.espOriginRoot=root
    end
    return UI.espOrigin
end
local function espColors()
    if settings.ESPColor=="Rosa" then return Color3.fromRGB(255,112,220),Color3.fromRGB(154,75,252) end
    if settings.ESPColor=="Dourado" then return Color3.fromRGB(255,207,76),Color3.fromRGB(255,133,65) end
    if settings.ESPColor=="Vermelho" then return Color3.fromRGB(255,91,100),Color3.fromRGB(255,154,80) end
    if settings.ESPColor=="Custom" then
        local custom=Color3.fromRGB(settings.ESPColorR,settings.ESPColorG,settings.ESPColorB)
        return custom,custom:Lerp(Color3.new(1,1,1),.34)
    end
    return C.ACCENT,C.ACCENT2
end
local function reportESPError(err)
    UI.espErrors+=1
    UI.espLastError=tostring(err):sub(1,150)
    warn("[Mayck Script ESP] "..UI.espLastError)
    if os.clock()-UI.espLastMessage>5 then
        UI.espLastMessage=os.clock()
        notify("ESP: falha ao desenhar efeito. Consulte a saída do jogo.",false)
    end
end
local function createESPModel(p,ch,root)
    local primary,secondary=espColors()
    local obj={Target=ch,Root=root,Skeleton={}}
    -- Registrar imediatamente permite limpar instâncias parcialmente criadas.
    espObjects[p]=obj
    local hi=Instance.new("Highlight")
    obj.Highlight=hi
    hi.Name="MayckESP_v54_Highlight"
    hi.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    hi.FillTransparency=0.65
    hi.OutlineTransparency=0
    hi.FillColor=secondary
    hi.OutlineColor=primary
    hi.Adornee=ch
    hi.Parent=ch

    local board=Instance.new("BillboardGui")
    obj.Billboard=board
    board.Name="MayckESP_v54_Billboard"
    board.Size=UDim2.fromOffset(184,56)
    board.Adornee=root
    board.StudsOffsetWorldSpace=Vector3.new(0,3,0)
    board.AlwaysOnTop=true
    board.MaxDistance=0
    board.LightInfluence=0
    -- Billboard no PlayerGui evita dependência do renderizador de GUI do Character.
    board.Parent=PlayerGui
    local frame=new("Frame",{Size=UDim2.fromScale(1,1),BackgroundTransparency=.15,
        BackgroundColor3=C.BG,BorderSizePixel=0},board)
    corner(frame,8)
    stroke(frame,primary,.2)
    local title=textLabel(frame,p.DisplayName,12,true)
    title.Size=UDim2.new(1,-14,0,21)
    title.Position=UDim2.fromOffset(7,2)
    local sub=textLabel(frame,"",10,false,"MUTED")
    sub.Size=UDim2.new(1,-14,0,17)
    sub.Position=UDim2.fromOffset(7,23)
    local back=new("Frame",{Position=UDim2.fromOffset(7,45),Size=UDim2.new(1,-14,0,6),
        BorderSizePixel=0,BackgroundColor3=C.BORDER},frame)
    corner(back,3)
    local fill=new("Frame",{Size=UDim2.fromScale(1,1),BorderSizePixel=0,
        BackgroundColor3=C.GOOD},back)
    corner(fill,3)
    obj.Title=title
    obj.Sub=sub
    obj.HealthBack=back
    obj.HealthFill=fill
    -- A ausência de um HRP local não impede que nomes e contornos sejam exibidos.
    obj.TargetAttachment=nil
    obj.Beam=nil
    obj.Box=nil
    return obj
end
local function buildESP(p)
    if destroyed or not settings.ESP or p==Me or p.Parent~=Players then return false end
    local ch=p.Character
    local target=ch and ch:FindFirstChild("HumanoidRootPart")
    if not target then
        UI.espUnready[p]=true
        return false
    end
    local old=espObjects[p]
    if old and old.Target==ch and old.Root==target and old.Billboard and old.Billboard.Parent
        and old.Highlight and old.Highlight.Parent then
        return true
    end
    clearESPPlayer(p)
    local ok,err=pcall(createESPModel,p,ch,target)
    if not ok then
        reportESPError(err)
        clearESPPlayer(p)
        UI.espUnready[p]=true
        return false
    end
    UI.espUnready[p]=nil
    return true
end
-- Efeitos opcionais independentes; não derrubam Highlight nem Billboard.
-- Adornments funcionam em R6/R15 sem usar Drawing API do executor.
local function ensureESPExtras(obj)
    if settings.ESPLines and not obj.Beam then
        local ok,err=pcall(function()
            local origin=originAttachment()
            if not origin then return end
            local target=new("Attachment",{Name="MayckESP_Target"},obj.Root)
            obj.TargetAttachment=target
            obj.Beam=new("Beam",{Name="MayckESP_Line",Attachment0=origin,
                Attachment1=target,FaceCamera=true,Width0=.045,Width1=.045,
                Transparency=NumberSequence.new(.18)},obj.Root)
        end)
        if not ok then
            reportESPError(err)
            safeDestroy(obj.Beam)
            safeDestroy(obj.TargetAttachment)
            obj.Beam=nil
            obj.TargetAttachment=nil
        end
    end
    if settings.ESPBox and not obj.Box then
        local ok,err=pcall(function()
            obj.Box=new("BoxHandleAdornment",{
                Name="MayckESP_Box",Adornee=obj.Root,AlwaysOnTop=true,
                Size=Vector3.new(4.5,6.5,2.8),Color3=C.ACCENT,
                Transparency=.2,Visible=false},obj.Root)
        end)
        if not ok then reportESPError(err) safeDestroy(obj.Box) obj.Box=nil end
    end
    if settings.ESPSkeleton and not obj.BonesCreated then
        local ch=obj.Target
        local pairsRig={
            {"Head","UpperTorso"},{"UpperTorso","LowerTorso"},
            {"UpperTorso","LeftUpperArm"},{"LeftUpperArm","LeftLowerArm"},
            {"LeftLowerArm","LeftHand"},{"UpperTorso","RightUpperArm"},
            {"RightUpperArm","RightLowerArm"},{"RightLowerArm","RightHand"},
            {"LowerTorso","LeftUpperLeg"},{"LeftUpperLeg","LeftLowerLeg"},
            {"LeftLowerLeg","LeftFoot"},{"LowerTorso","RightUpperLeg"},
            {"RightUpperLeg","RightLowerLeg"},{"RightLowerLeg","RightFoot"}
        }
        if not ch:FindFirstChild("UpperTorso") then
            pairsRig={{"Head","Torso"},{"Torso","Left Arm"},{"Torso","Right Arm"},
                {"Torso","Left Leg"},{"Torso","Right Leg"}}
        end
        local available=0
        for _,names in ipairs(pairsRig) do
            local limbA=ch:FindFirstChild(names[1])
            local limbB=ch:FindFirstChild(names[2])
            if limbA and limbB and limbA:IsA("BasePart") and limbB:IsA("BasePart") then
                local ok,err=pcall(function()
                    local handle=new("BoxHandleAdornment",{
                        Name="MayckESP_SkeletonBone",Adornee=obj.Root,
                        AlwaysOnTop=true,Size=Vector3.new(.12,1,.12),
                        Color3=C.ACCENT,Transparency=.04,Visible=false},obj.Root)
                    table.insert(obj.Skeleton,{A=limbA,B=limbB,Adornment=handle})
                    available+=1
                end)
                if not ok then reportESPError(err) end
            end
        end
        obj.BonesCreated=available>0
    end
end

-- Caixa com dimensões reais do corpo; acompanha altura e acessórios básicos.
local function syncESPBox(obj,visible,color)
    if not obj.Box then return end
    obj.Box.Visible=visible and settings.ESPBox
    if not obj.Box.Visible then return end
    obj.Box.Color3=color
    local boundingCF,boundingSize=obj.Target:GetBoundingBox()
    obj.Box.Size=boundingSize+Vector3.new(.18,.18,.18)
    obj.Box.CFrame=obj.Root.CFrame:ToObjectSpace(boundingCF)
end

-- Segmentos ficam acima das paredes e não dependem do pipeline Beam.
local function syncESPSkeleton(obj,visible,color)
    for _,bone in ipairs(obj.Skeleton) do
        local box=bone.Adornment
        if box and box.Parent then
            local active=visible and settings.ESPSkeleton and bone.A.Parent and bone.B.Parent
            box.Visible=not not active
            if active then
                local a=bone.A.Position
                local b=bone.B.Position
                local delta=b-a
                local len=delta.Magnitude
                if len>.02 then
                    local middle=(a+b)*.5
                    local worldCF=CFrame.lookAt(middle,b)*CFrame.Angles(math.pi/2,0,0)
                    box.CFrame=obj.Root.CFrame:ToObjectSpace(worldCF)
                    box.Size=Vector3.new(.13,len,.13)
                    box.Color3=color
                else
                    box.Visible=false
                end
            end
        end
    end
end

    for p in pairs(espObjects) do clearESPPlayer(p) end

local function updateESP()
    if not settings.ESP then return end
    if os.clock()-UI.espLastRefresh>=1 then
        UI.espLastRefresh=os.clock()
        for _,p in ipairs(Players:GetPlayers()) do
            if p~=Me then buildESP(p) end
        end
    end
    local localRoot=myRoot()
    -- Roblox limita Highlights simultâneos; prioriza os mais próximos.
    -- Billboard e 3D Box funcionam mesmo fora das vagas de Highlight.
    local nearest={}
    local shown,total=0,0
    for p,obj in pairs(espObjects) do
        if p.Parent~=Players or p.Character~=obj.Target or not obj.Root.Parent then
            clearESPPlayer(p)
        else
            total+=1
            local distance=localRoot and (obj.Root.Position-localRoot.Position).Magnitude or 0
            local sameTeam=Me.Team~=nil and p.Team~=nil and p.Team==Me.Team
            local visible=distance<=settings.MaxDistance and
                not (settings.ESPTeamFilter and sameTeam)
            obj.Visible=visible
            obj.Distance=distance
            if visible then
                shown+=1
                nearest[#nearest+1]={Player=p,Distance=distance}
            end
        end
    end
    table.sort(nearest,function(a,b) return a.Distance<b.Distance end)
    local highlighted={}
    for i=1,math.min(#nearest,25) do highlighted[nearest[i].Player]=true end

    for p,obj in pairs(espObjects) do
        local ok,err=pcall(function()
            local visible=obj.Visible
            local distance=obj.Distance or 0
            obj.Highlight.Enabled=visible and highlighted[p]==true
            obj.Billboard.Enabled=visible and
                (settings.ESPName or settings.ESPDistance or settings.ESPHealth)
            obj.Title.Visible=settings.ESPName
            obj.Title.Text=p.DisplayName
            obj.Sub.Visible=settings.ESPDistance
            obj.Sub.Text=string.format("@%s • %d studs",p.Name,math.floor(distance))
            obj.HealthBack.Visible=settings.ESPHealth
            local h=obj.Target:FindFirstChildOfClass("Humanoid")
            if h then
                local portion=math.clamp(h.Health/math.max(h.MaxHealth,1),0,1)
                obj.HealthFill.Size=UDim2.fromScale(portion,1)
                obj.HealthFill.BackgroundColor3=Color3.fromRGB(
                    math.floor(255*(1-portion)),math.floor(255*portion),95)
            else
                obj.HealthFill.Size=UDim2.fromScale(0,1)
            end
            local primary,secondary=espColors()
            obj.Highlight.FillColor=secondary
            obj.Highlight.OutlineColor=primary
        end)
        if not ok then
            reportESPError(err)
            clearESPPlayer(p)
        else
            local optionalOK,optionalError=pcall(function()
                local visible=obj.Visible
                local primary=espColors()
                if settings.ESPLines or settings.ESPBox or settings.ESPSkeleton then
                    ensureESPExtras(obj)
                end
                if obj.Beam then
                    obj.Beam.Color=ColorSequence.new(primary)
                    local origin=originAttachment()
                    if origin then obj.Beam.Attachment0=origin end
                    obj.Beam.Enabled=visible and settings.ESPLines and origin~=nil
                end
                syncESPBox(obj,visible,primary)
                syncESPSkeleton(obj,visible,primary)
            end)
            if not optionalOK then
                -- Erro no efeito opcional não remove o nome/vida/Highlight.
                reportESPError(optionalError)
            end
        end
    end
    if V4.espStatus and V4.espStatus.Parent and os.clock()-UI.espLastStatus>.35 then
        UI.espLastStatus=os.clock()
        V4.espStatus.Text=string.format(
            "ESP: %d visíveis • %d preparados • %d Highlights • %d studs",
            shown,total,math.min(#nearest,25),settings.MaxDistance)
        if UI.espLastError and UI.espErrors>0 then
            V4.espStatus.Text=V4.espStatus.Text.." • erros: "..UI.espErrors
        end
    end
end

    if p==Me or watchedPlayers[p] then return end
    watchedPlayers[p]=p.CharacterAdded:Connect(function()

            if not destroyed and settings.ESP and p.Parent==Players then buildESP(p) end
        end)
    end)
    if settings.ESP then buildESP(p) end
end

    if spectateTarget==p then spectateTarget=nil cameraReset() end
end

featureChanged=function(key)
    if destroyed then return end
    if key=="ESP" then
        if settings.ESP then
            for _,p in ipairs(Players:GetPlayers()) do if p~=Me then buildESP(p) end end
        else clearESP() end
        if V4.espQuickButton then
            V4.espQuickButton.Text=settings.ESP and "DESATIVAR ESP • ligado" or "ATIVAR ESP • destaque dos jogadores"
        end
        if V4.espStatus then
            V4.espStatus.Text=settings.ESP and "ESP ligado • procurando personagens..." or "ESP desligado. Ative ESP geral e ajuste Box/Skeleton na aba Visual."
        end
    elseif key=="Speed" or key=="SpeedValue" or key=="AutoSprint" or key=="SprintValue" then
        setSpeed()
    elseif key=="JumpBoost" or key=="JumpPower" then
        setJumpPower()
    elseif key=="Fly" then
        if settings.Fly then
            if settings.Freecam then
                settings.Freecam=false
                stopFreecam()
                if toggles.Freecam then toggles.Freecam.paint() end
            end
            startFly()
        else stopFly() end
        UI.MobileFly.Visible=settings.Fly and UserInputService.TouchEnabled
    elseif key=="Noclip" then
        if not settings.Noclip then restoreNoclip() end
    elseif key=="Freecam" then
        if settings.Freecam then
            if settings.Fly then
                settings.Fly=false
                stopFly()
                UI.MobileFly.Visible=false
                if toggles.Fly then toggles.Fly.paint() end
            end
            startFreecam()
            togglePanel(false)
            notify("Freecam: botão direito + mouse, WASD, Q/Ctrl.",true)
        else stopFreecam() end
    elseif key=="CameraFOV" then
        local cam=Workspace.CurrentCamera
        if cam then cam.FieldOfView=settings.CameraFOV end
    elseif key=="Compact" then
        fitToScreen()
    elseif key=="Performance" then
        V4.setPerformance()
    elseif key=="Radar" then
        V4.updateRadar()
    elseif key=="WalkOnWater" and not settings.WalkOnWater then
        V4.stepWater()
    elseif key=="PanelTransparency" then
        Root.BackgroundTransparency=settings.PanelTransparency/100
    elseif key=="ESPColorR" or key=="ESPColorG" or key=="ESPColorB" then
        settings.ESPColor="Custom"
        UI.espColorBtn.Text="RGB"
    end
end

local function cleanup()
    if destroyed then return end
    -- The Park Hero engine owns a separate CoreGui. Shut it down BEFORE
    -- disconnecting the main GUI events, otherwise its keys/buttons survive.
    if UI.Park and UI.Park.engine then
        local engine = UI.Park.engine
        UI.Park.engine = nil
        pcall(function() engine.Pause() end)
        pcall(function() engine.Shutdown() end)
    end
    -- Defensive cleanup of an orphaned Park Hero ScreenGui from an old run.
    local oldParkGui = game:GetService("CoreGui"):FindFirstChild("MayckParkHeroIntegratedEngine")
    if oldParkGui then pcall(function() oldParkGui:Destroy() end) end
    if UI.hiddenMois7Gui and UI.hiddenMois7Gui.Parent then
        pcall(function() UI.hiddenMois7Gui.Enabled=true end)
        UI.hiddenMois7Gui=nil
    end
    -- Execute desligamento enquanto as rotinas ainda podem restaurar estado.
    disableAll()
    destroyed=true
    for p in pairs(watchedPlayers) do unwatchPlayer(p) end
    clearESP()
    if V4.waterPlatform then V4.waterPlatform:Destroy() V4.waterPlatform=nil end
    for _,d in pairs(V4.radarDots) do if d then d:Destroy() end end
    for _,c in ipairs(connections) do pcall(function() c:Disconnect() end) end
    table.clear(connections)
    if Gui and Gui.Parent then Gui:Destroy() end
end
UI.destroyBtn.MouseButton1Click:Connect(cleanup)
Gui.Destroying:Connect(function()
    if not destroyed then cleanup() end
end)

connect(UserInputService.InputBegan,function(input,processed)
    if destroyed then return end
    if input.UserInputType==Enum.UserInputType.MouseButton2 then UI.isRightMouse=true end
    if processed or UserInputService:GetFocusedTextBox() then return end
    if input.KeyCode==Enum.KeyCode.RightAlt then
        disableAll()
        return
    end
    if input.KeyCode==settings.Hotkey then togglePanel() return end
    if input.KeyCode==settings.DashHotkey then V4.performDash() end
    V4.tryClickTeleport(input,processed)
end)
connect(UserInputService.InputEnded,function(input)
    if input.UserInputType==Enum.UserInputType.MouseButton2 then UI.isRightMouse=false end
end)
connect(UserInputService.InputChanged,function(input,processed)
    if destroyed or not settings.Freecam then return end
    if processed or UserInputService:GetFocusedTextBox() then return end
    if input.UserInputType==Enum.UserInputType.MouseMovement and UI.isRightMouse then
        UI.fcYaw-=input.Delta.X*0.003
        UI.fcPitch=math.clamp(UI.fcPitch-input.Delta.Y*0.003,-1.5,1.5)
    elseif input.UserInputType==Enum.UserInputType.Touch and not isOpen then
        UI.fcYaw-=input.Delta.X*0.003
        UI.fcPitch=math.clamp(UI.fcPitch-input.Delta.Y*0.003,-1.5,1.5)
    end
end)

connect(RunService.Heartbeat,function(dt)
    if destroyed then return end
    local root=myRoot()
    local hum=myHumanoid()
    if settings.AntiVoid and root then
        if root.Position.Y > -15 and hum and hum.FloorMaterial~=Enum.Material.Air
            and not settings.Fly and not settings.Noclip then
            UI.antiVoidSaved=root.CFrame
        end
        if root.Position.Y < -70 and os.clock()-UI.lastVoidTime > 1.5 then
            UI.lastVoidTime=os.clock()
            if UI.antiVoidSaved then root.CFrame=UI.antiVoidSaved*CFrame.new(0,4,0)
            else root.CFrame=CFrame.new(root.Position.X,20,root.Position.Z) end
            root.AssemblyLinearVelocity=Vector3.zero
        end
    end
    if settings.Speed or settings.AutoSprint or UI.speedSavedHumanoid then
        setSpeed()
    end
    if settings.BunnyHop and hum and hum.Health>0 and hum.MoveDirection.Magnitude>.1
        and hum.FloorMaterial~=Enum.Material.Air and os.clock()-V4.lastBunny>.22 then
        V4.lastBunny=os.clock()
        hum.Jump=true
    end
    if settings.WalkOnWater and os.clock()-V4.lastWaterScan>.12 then
        V4.lastWaterScan=os.clock()
        V4.stepWater()
    end
end)

connect(RunService.RenderStepped,function(dt)
    if destroyed then return end
    if settings.Fly then flyMove(dt) end
    if settings.Freecam then stepFreecam(dt) end
    if spectateTarget and not settings.Freecam then
        local cam=Workspace.CurrentCamera
        local hum=spectateTarget.Character and spectateTarget.Character:FindFirstChildOfClass("Humanoid")
        if cam and hum and cam.CameraSubject~=hum then cam.CameraSubject=hum end
    end
    UI.radarElapsed+=dt
    if UI.radarElapsed>.25 then
        UI.radarElapsed=0
        local ok,err=pcall(V4.updateRadar)
        if not ok then warn("[Mayck Radar] "..tostring(err)) end
    end
    UI.espElapsed+=dt
    if UI.espElapsed>0.15 then
        UI.espElapsed=0
        local ok,err=pcall(updateESP)
        if not ok then warn("[Mayck ESP] Atualização: "..tostring(err)) end
    end
    UI.fpsElapsed+=dt
    UI.frameCount+=1
    if UI.fpsElapsed>=1 then
        local fps=math.floor(UI.frameCount/UI.fpsElapsed+0.5)
        local ping="--"
        pcall(function() ping=tostring(math.floor(Me:GetNetworkPing()*1000+0.5)) end)
        UI.fpsLabel.Text=string.format("FPS %d  |  PING %s ms",fps,ping)
        UI.fpsElapsed=0
        UI.frameCount=0
        UI.serverInfo.Text=string.format("Jogadores: %d/%d  •  PlaceId: %s  •  Server: %s",
            #Players:GetPlayers(),Players.MaxPlayers,tostring(game.PlaceId),
            game.JobId~="" and game.JobId:sub(1,8).."..." or "Studio")
        local uptime=math.floor(os.clock()-UI.sessionStartClock)
        local myPos=myRoot()
        local coords=myPos and string.format("X %.0f • Y %.0f • Z %.0f",
            myPos.Position.X,myPos.Position.Y,myPos.Position.Z) or "Personagem indisponível"
        UI.sessionStatus.Text=string.format("Sessão: %02d:%02d  |  %s",
            math.floor(uptime/60),uptime%60,coords)
        local v=viewport()
        if v~=UI.previousVP then
            UI.previousVP=v
            fitToScreen()
            local floatCenter=UI.FloatButton.AbsolutePosition+UI.FloatButton.AbsoluteSize/2
            local fp=V4.clampFloatPosition(floatCenter)
            UI.FloatButton.Position=UDim2.fromOffset(fp.X,fp.Y)
        end
    end
end)
